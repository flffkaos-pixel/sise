본 정보는 투자 참고용이며, 투자 판단과 책임은 이용자에게 있습니다. 과거 데이터 기반 예측은 실제 결과와 다를 수 있습니다.

document.getElementById('contactForm').addEventListener('submit', async (e) => { e.preventDefault(); const msgEl = document.getElementById('formMsg'); msgEl.className = 'msg'; msgEl.textContent = '전송 중...'; msgEl.style.display = 'block'; const formData = new FormData(e.target); const data = Object.fromEntries(formData); try { const res = await fetch('/api/contact', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) }); const result = await res.json(); if (res.ok) { msgEl.className = 'msg success'; msgEl.textContent = '문의가 접수되었습니다. 영업일 기준 1~2일 내 답변 드리겠습니다.'; e.target.reset(); } else { msgEl.className = 'msg error'; msgEl.textContent = result.error || '전송 실패. 다시 시도해 주세요.'; } } catch (err) { msgEl.className = 'msg error'; msgEl.textContent = '네트워크 오류. 잠시 후 다시 시도해 주세요.'; } });
